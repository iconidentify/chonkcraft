# Project-local backend for the native decision micro-oracle.
#
# Installed into tools/bne-harness/.venv-micro-oracle by
# scripts/micro-oracle-venv.sh; never into the machine's Python. Both pins are
# exact because a replay whose backend drifted is not a reproduction.
#
# unicorn 2.1.3 is the CPU emulator. 2.1.1 aborts with SIGBUS the moment an
# x86-32 context is created on Apple Silicon under Python 3.14, so the pin is
# load-bearing rather than tidy.
unicorn==2.1.3
# capstone decodes the executed instructions so a branch can be told from a
# move. 5.0.1 imports distutils, which Python 3.14 removed.
capstone==5.0.7
